async function inicializarApp() { try { // Tenta inicializar o LIFF await liff.init({ liffId: liffId }); if (liff.isLoggedIn()) { const profile = await liff.getProfile(); localStorage.setItem('line_user_id', profile.userId); localStorage.setItem('line_user_name', profile.displayName); liberarConteudoApp(); } else { // Se não estiver logado no LIFF, mas já salvou antes, libera if (localStorage.getItem('line_user_id')) { liberarConteudoApp(); } else { document.getElementById('login-overlay').style.display = 'flex'; } } } catch (err) { console.warn("Erro no LIFF (normal se testando fora do app do LINE):", err); // Se houver falha no LIFF (ex: abrindo no Chrome/Safari), libera o uso local para testes if (localStorage.getItem('line_user_id')) { liberarConteudoApp(); } else { // Cria um usuário de teste temporário para não travar sua tela localStorage.setItem('line_user_id', 'usuario_teste_local'); liberarConteudoApp(); } } } function liberarConteudoApp() { document.getElementById('login-overlay').style.display = 'none'; document.getElementById('app-content').style.display = 'block'; } function fazerLoginLine() { try { if (!liff.isLoggedIn()) { liff.login(); } else { liberarConteudoApp(); } } catch (e) { // Fallback caso o liff.login falhe fora do app localStorage.setItem('line_user_id', 'usuario_teste_local'); liberarConteudoApp(); } }
