function desenharAcessorios() { const camada = document.getElementById( "acessorio-camada" ); camada.innerHTML = ""; /* * MODO DE CONFIGURAÇÃO: * mostra SOMENTE o acessório * selecionado no painel. * * Assim nenhum acessório que já * esteja equipado atrapalha * o posicionamento. */ const selecionado = obterSelecionado(); if ( !selecionado || !ACESSORIOS[selecionado] ) { return; } const produto = ACESSORIOS[selecionado]; const elemento = criarVisualAcessorio( selecionado, produto ); const ajuste = obterAjuste( selecionado ); elemento.style.setProperty( "--x", ajuste.x + "px" ); elemento.style.setProperty( "--y", ajuste.y + "px" ); elemento.style.setProperty( "--scale", ajuste.scale ); elemento.style.setProperty( "--rot", ajuste.rot + "deg" ); elemento.style.zIndex = "200"; camada.appendChild( elemento ); }
