// Certifique-se de que o firebase/firestore já foi importado ou inicializado no seu projeto // Exemplo usando a modular SDK do Firebase (v9+): import { initializeApp } from "https://www.gstatic.com/firebasejs/10.x.x/firebase-app.js"; import { getFirestore, collection, query, where, orderBy, limit, getDocs } from "https://www.gstatic.com/firebasejs/10.x.x/firebase-firestore.js"; // (Se você já inicializou o db em outro arquivo, basta importá-lo aqui) // const db = getFirestore(app); async function carregarRankingJogo(idJogo) { const listContainer = document.getElementById('rankingList'); listContainer.innerHTML = '
Carregando Top 100 do Firebase...
'; try { let dadosRanking = []; // OPÇÃO A: Se você salva todos os scores numa coleção única chamada 'rankings' filtrando pelo jogo: const q = query( collection(db, "rankings"), where("jogo", "==", idJogo), orderBy("pontos", "desc"), limit(100) ); /* OPÇÃO B: Se você cria uma coleção para cada jogo (ex: 'ranking_minicraque', 'ranking_velha'): const q = query( collection(db, `ranking_${idJogo}`), orderBy("pontos", "desc"), limit(100) ); */ const querySnapshot = await getDocs(q); querySnapshot.forEach((doc) => { const data = doc.data(); dadosRanking.push({ nome: data.nome || "Jogador Anônimo", pontos: Number(data.pontos) || 0 }); }); // Se o banco estiver vazio para esse jogo, exibe o aviso if (dadosRanking.length === 0) { listContainer.innerHTML = '
Nenhuma pontuação registrada ainda para este jogo. Seja o primeiro!
'; return; } // Monta o HTML do Top 100 com os dados reais do Firebase let html = ''; dadosRanking.forEach((jogador, index) => { let posicao = index + 1; let podiumClass = ''; let iconePosicao = `#${posicao}`; if (posicao === 1) { podiumClass = 'podium-1'; iconePosicao = '🥇'; } else if (posicao === 2) { podiumClass = 'podium-2'; iconePosicao = '🥈'; } else if (posicao === 3) { podiumClass = 'podium-3'; iconePosicao = '🥉'; } html += `
${iconePosicao} ${escapeHtml(jogador.nome)}
⭐ ${jogador.pontos.toLocaleString()} pts
`; }); listContainer.innerHTML = html; } catch (error) { console.error("Erro ao buscar ranking do Firebase:", error); listContainer.innerHTML = '
Erro ao carregar o ranking. Verifique sua conexão com o Firebase.
'; } }
