// Tradução integrada com o backend seguro (/api/traduzir) async function traduzirTexto() { const texto = document.getElementById('textoInput').value.trim(); const box = document.getElementById('resultadoBox'); if (!texto) { alert('Digite algum texto primeiro!'); return; } box.style.display = 'block'; box.innerText = 'Traduzindo...'; try { // Envia o texto para a rota backend na pasta api/traduzir.js const resposta = await fetch('/api/traduzir', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ texto }) }); const dados = await resposta.json(); if (resposta.ok && dados && dados.traducao) { let traducao = dados.traducao; const contemJapones = /[\u3040-\u30ff\u4e00-\u9faf]/.test(texto); // Se a tradução for para japonês, adiciona o Romaji limpo ao lado entre parênteses if (!contemJapones && typeof wanakana !== 'undefined') { let textoLimpo = traducao.replace(/[.!?。、]/g, ''); let romaji = wanakana.toRomaji(textoLimpo); if (romaji) { romaji = romaji.charAt(0).toUpperCase() + romaji.slice(1); } if (romaji && romaji.toLowerCase() !== traducao.toLowerCase()) { box.innerHTML = `${traducao} (${romaji})`; } else { box.innerText = traducao; } } else { box.innerText = traducao; } } else { box.innerText = dados.erro || 'Erro ao processar a tradução.'; } } catch (e) { box.innerText = 'Erro ao conectar com o servidor.'; } }
